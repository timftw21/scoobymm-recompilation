cmake_minimum_required(VERSION 3.24)
project(native_recompiler LANGUAGES C CXX)

include(FetchContent)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(ZYDIS_FEATURE_ENCODER OFF)
set(ZYDIS_FEATURE_FORMATTER OFF)
set(ZYDIS_BUILD_EXAMPLES OFF)
set(ZYDIS_BUILD_TOOLS OFF)
set(ZYDIS_BUILD_TESTS OFF)
FetchContent_Declare(zydis
    GIT_REPOSITORY https://github.com/zyantific/zydis.git
    GIT_TAG a2278f1d254e492f6a6b39f6cb5d1f5d515659dc
    GIT_SUBMODULES dependencies/zycore)
FetchContent_MakeAvailable(zydis)

set(target_registry "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/supported_target.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${target_registry}")
file(READ "${target_registry}" registry)
foreach(field IN ITEMS file_size file_sha256 entry_point kernel_thunks)
    string(JSON target_${field} GET "${registry}" targets 0 ${field})
endforeach()
configure_file(src/xbox/target.h.in generated/target.h @ONLY)

string(JSON import_count LENGTH "${registry}" targets 0 kernel_imports)
set(kernel_cases "")
set(import_ordinals "")
math(EXPR import_last "${import_count} - 1")
foreach(index RANGE 0 ${import_last})
    string(JSON ordinal GET "${registry}" targets 0 kernel_imports ${index} ordinal)
    if(NOT ordinal IN_LIST import_ordinals)
        string(JSON name GET "${registry}" targets 0 kernel_imports ${index} name)
        string(APPEND kernel_cases "    case ${ordinal}: return \"${name}\";\n")
        list(APPEND import_ordinals "${ordinal}")
    endif()
endforeach()
configure_file(src/xbox/kernel_names.h.in generated/kernel_names.h @ONLY)
set(recovery_index "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/boot-recovery.tsv")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${recovery_index}")
file(STRINGS "${recovery_index}" recovery_rows)
list(POP_FRONT recovery_rows)
set(recovery_entries "")
foreach(row IN LISTS recovery_rows)
    if(NOT row MATCHES "^(0x[0-9A-F]+)\t(0x[0-9A-F]+)\t(0x[0-9A-F]+)\t(0x[0-9A-F]+)$")
        message(FATAL_ERROR "Invalid boot recovery range ${row}")
    endif()
    string(APPEND recovery_entries "    {${CMAKE_MATCH_1}, ${CMAKE_MATCH_2}, ${CMAKE_MATCH_3}, ${CMAKE_MATCH_4}},\n")
endforeach()
configure_file(src/xbox/boot_recovery.h.in generated/boot_recovery.h @ONLY)
set(platform_index "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/platform-bindings.tsv")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${platform_index}")
file(STRINGS "${platform_index}" platform_rows)
list(POP_FRONT platform_rows)
set(platform_enum "")
set(platform_entries "")
foreach(row IN LISTS platform_rows)
    if(NOT row MATCHES "^([a-zA-Z][a-zA-Z0-9_]+)\t(0x[0-9A-F]+)\t([0-9A-F]+)$")
        message(FATAL_ERROR "Invalid platform binding ${row}")
    endif()
    string(APPEND platform_enum "    ${CMAKE_MATCH_1} = ${CMAKE_MATCH_2},\n")
    string(APPEND platform_entries "    {PlatformApi::${CMAKE_MATCH_1}, \"${CMAKE_MATCH_1}\", \"${CMAKE_MATCH_3}\"},\n")
endforeach()
configure_file(src/xbox/platform_bindings.h.in generated/platform_bindings.h @ONLY)

set(module_index "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/boot_modules.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${module_index}")
file(READ "${module_index}" modules_json)
string(JSON verified_disc_sha256 GET "${modules_json}" disc_sha256)
string(JSON module_count LENGTH "${modules_json}" modules)
set(module_entries "")
if(module_count GREATER 0)
    math(EXPR module_last "${module_count} - 1")
    foreach(index RANGE 0 ${module_last})
        foreach(field IN ITEMS name disc_path record_file_offset record_size record_sha256 object_record_offset object_size object_sha256)
            string(JSON module_${field} GET "${modules_json}" modules ${index} ${field})
        endforeach()
        string(JSON module_registration GET "${modules_json}" modules ${index} registration section_offset)
        string(JSON module_call GET "${modules_json}" modules ${index} registration xbe_call_address)
        string(APPEND module_entries "    {\"${module_name}\", \"${module_disc_path}\", ${module_record_file_offset}, ${module_record_size}, \"${module_record_sha256}\", ${module_object_record_offset}, ${module_object_size}, \"${module_object_sha256}\", ${module_registration}, ${module_call}},\n")
    endforeach()
endif()
configure_file(src/xbox/boot_modules.h.in generated/boot_modules.h @ONLY)

add_library(xbox-media STATIC src/xbox/disc.cpp src/xbox/file.cpp src/xbox/xbe.cpp)
target_compile_features(xbox-media PUBLIC cxx_std_23)
target_include_directories(xbox-media PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties(xbox-media PROPERTIES CXX_EXTENSIONS OFF)
add_executable(native-recompiler src/recompile.cpp src/xbox/batch.cpp src/xbox/call_sources.cpp src/xbox/call_evidence.cpp src/xbox/lower.cpp src/xbox/vertex_loop.cpp)
target_compile_features(native-recompiler PRIVATE cxx_std_23)
set_target_properties(native-recompiler PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(native-recompiler PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(native-recompiler PRIVATE Zydis xbox-media)
if(WIN32)
    target_compile_definitions(xbox-media PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(xbox-media PRIVATE bcrypt)
else()
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(xbox-media PRIVATE OpenSSL::Crypto)
endif()
if(MSVC)
    target_compile_options(native-recompiler PRIVATE /W4)
    target_compile_options(xbox-media PRIVATE /W4)
else()
    target_compile_options(native-recompiler PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(xbox-media PRIVATE -Wall -Wextra -Wpedantic)
endif()

set(XBOX_GENERATED_SOURCE "" CACHE PATH "Directory containing the generated Xbox C++ batch")
if(XBOX_GENERATED_SOURCE)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The emitted assembly helpers currently target Windows x64")
    endif()
    enable_language(ASM_MASM)
    file(GLOB generated_shards CONFIGURE_DEPENDS "${XBOX_GENERATED_SOURCE}/game.*.cpp")
    add_library(xbox-recompiled STATIC
        "${XBOX_GENERATED_SOURCE}/game.cpp" "${XBOX_GENERATED_SOURCE}/game.asm"
        ${generated_shards} src/xbox/cpu.cpp src/xbox/floating.cpp src/xbox/cpu_floating.asm)
    target_compile_features(xbox-recompiled PRIVATE cxx_std_23)
    target_include_directories(xbox-recompiled PUBLIC src "${XBOX_GENERATED_SOURCE}")
    find_package(SDL3 3.4 CONFIG QUIET)
    if(NOT SDL3_FOUND)
        set(SDL_SHARED OFF)
        set(SDL_STATIC ON)
        foreach(option IN ITEMS SDL_TEST_LIBRARY SDL_TESTS SDL_EXAMPLES SDL_INSTALL SDL_GPU SDL_RENDER SDL_OPENGL SDL_OPENGLES SDL_VULKAN)
            set(${option} OFF)
        endforeach()
        FetchContent_Declare(sdl3
            URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-3.4.20
            URL_HASH SHA256=5CB56D4223A41283F3C2BAF6F1263FBA171692409AE405A50417A7C8E7E28DEB
            DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
        FetchContent_MakeAvailable(sdl3)
    endif()
    FetchContent_Declare(imgui
        URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.93.0
        URL_HASH SHA256=F7BF6BC76E5D79D40D751096663E6C5D18DAF08129DD52E7F7B09E4251C8EEEE
        DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
    FetchContent_MakeAvailable(imgui)
    include(cmake/imgui-shaders.cmake)
    add_library(scoobymm-imgui STATIC
        "${imgui_SOURCE_DIR}/imgui.cpp" "${imgui_SOURCE_DIR}/imgui_draw.cpp"
        "${imgui_SOURCE_DIR}/imgui_tables.cpp" "${imgui_SOURCE_DIR}/imgui_widgets.cpp"
        "${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/imgui_impl_dx11.cpp")
    target_compile_features(scoobymm-imgui PRIVATE cxx_std_23)
    target_include_directories(scoobymm-imgui PRIVATE src/xbox)
    target_include_directories(scoobymm-imgui PUBLIC "${imgui_SOURCE_DIR}" "${imgui_SOURCE_DIR}/backends")
    target_link_libraries(scoobymm-imgui PRIVATE SDL3::SDL3)
    find_package(Git REQUIRED)
    find_program(CARGO_EXECUTABLE cargo REQUIRED)
    FetchContent_Declare(dsp56300
        GIT_REPOSITORY https://github.com/xemu-project/dsp56300.git
        GIT_TAG aab3649fc452daf03da8b5e36fdc0c308d874101
        GIT_SUBMODULES ""
        UPDATE_DISCONNECTED TRUE
        PATCH_COMMAND "${GIT_EXECUTABLE}" apply "${CMAKE_CURRENT_SOURCE_DIR}/cmake/dsp56300-return.patch")
    FetchContent_MakeAvailable(dsp56300)
    set(dsp_library "${dsp56300_BINARY_DIR}/release/dsp56300_emu_ffi.lib")
    add_custom_command(OUTPUT "${dsp_library}"
        COMMAND "${CARGO_EXECUTABLE}" build --locked --release --jobs 2 -p dsp56300-emu-ffi
                --manifest-path "${dsp56300_SOURCE_DIR}/Cargo.toml" --target-dir "${dsp56300_BINARY_DIR}"
        DEPENDS "${dsp56300_SOURCE_DIR}/Cargo.lock" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/dsp56300-return.patch"
        COMMENT "Building DSP56300 audio processor" VERBATIM)
    add_custom_target(scoobymm-dsp DEPENDS "${dsp_library}")
    add_library(xbox-runtime STATIC
        src/xbox/settings.cpp src/xbox/settings_menu.cpp
        src/xbox/boot.cpp src/xbox/audio.cpp src/xbox/audio_effects.cpp src/xbox/guest_audio.cpp
        src/xbox/guest_graphics.cpp src/xbox/gpu.cpp src/xbox/render.cpp src/xbox/shaders.cpp
        src/xbox/vertex_program.cpp src/xbox/play_window.cpp src/xbox/play_diagnostics.cpp)
    target_compile_features(xbox-runtime PUBLIC cxx_std_23)
    add_dependencies(xbox-runtime scoobymm-dsp)
    target_include_directories(xbox-runtime PRIVATE "${dsp56300_SOURCE_DIR}/crates/emu-ffi/include")
    target_link_libraries(xbox-runtime PRIVATE "${dsp_library}" ws2_32 userenv bcrypt ntdll)
    target_include_directories(xbox-runtime PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_options(xbox-runtime PRIVATE /W4)
    set_target_properties(xbox-runtime PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries(xbox-runtime PRIVATE xbox-media xbox-recompiled SDL3::SDL3 scoobymm-imgui d3d11 dxgi windowscodecs ole32 advapi32 comdlg32)
    target_compile_definitions(native-recompiler PRIVATE XBOX_OFFLINE_BOOT=1)
    target_sources(native-recompiler PRIVATE src/xbox/stream_check.cpp src/xbox/shader_compiler.cpp)
    target_link_libraries(native-recompiler PRIVATE xbox-runtime d3dcompiler)
    enable_language(RC)
    set(XBOX_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/data/local/default.xbe" CACHE FILEPATH "Verified Xbox executable for material shader generation")
    set(shader_bank "${CMAKE_CURRENT_BINARY_DIR}/generated/scoobymm-shaders.bin")
    add_custom_command(OUTPUT "${shader_bank}"
        COMMAND native-recompiler bake-shaders "${shader_bank}" "${CMAKE_CURRENT_BINARY_DIR}/shader-cache" "${XBOX_EXECUTABLE}"
        DEPENDS native-recompiler "${XBOX_EXECUTABLE}"
        COMMENT "Compiling renderer and Xbox material shaders"
        VERBATIM)
    add_custom_target(scoobymm-shaders DEPENDS "${shader_bank}")
    add_executable(scoobymm src/play.cpp src/xbox/shader_bank.cpp resources/scoobymm.rc)
    add_dependencies(scoobymm scoobymm-shaders)
    set_source_files_properties(resources/scoobymm.rc PROPERTIES
        OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/scoobymm.ico;${shader_bank}")
    target_include_directories(scoobymm PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_options(scoobymm PRIVATE /W4)
    target_link_libraries(scoobymm PRIVATE xbox-runtime)
    set_target_properties(scoobymm PROPERTIES CXX_EXTENSIONS OFF)
endif()
